313# endif

320 *str_ptr += __builtin_ctzll(qw.dw[0]) / 8;
321 else if (qw.dw[1])
322 *str_ptr += 8 + __builtin_ctzll(qw.dw[1]) / 8;
323 else {
324 *str_ptr += 16;

326 prev_data = orig_data;
327#endif

329 }

343#endif

348#endif

350 }
